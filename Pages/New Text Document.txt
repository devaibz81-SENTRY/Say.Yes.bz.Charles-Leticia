<script>
  const stage = document.getElementById('stage');
  const glowWash = document.getElementById('glowWash');
  const canvas = document.getElementById('fwCanvas');
  const ctx = canvas.getContext('2d');

  // refined palette: warm gold, champagne, soft ember, icy blue-white
  const palette = ['#e8c27a', '#fff3d0', '#ff9d6c', '#bfe3ff', '#f4e3c1', '#ffd9a0'];

  function rand(min, max){ return Math.random() * (max - min) + min; }
  function choice(arr){ return arr[Math.floor(Math.random() * arr.length)]; }

  let W, H, DPR;
  function resize(){
    DPR = Math.min(window.devicePixelRatio || 1, 2);
    W = canvas.clientWidth;
    H = canvas.clientHeight;
    canvas.width = W * DPR;
    canvas.height = H * DPR;
    ctx.setTransform(DPR, 0, 0, DPR, 0, 0);
  }
  window.addEventListener('resize', resize);
  resize();

  // --- physics constants ---
  const GRAVITY = 55;          // px/s^2, pulls everything downward
  const DRAG = 0.985;          // per-frame velocity retention (air resistance)
  const WIND = () => Math.sin(Date.now() / 4000) * 4; // gentle drifting breeze

  const rockets = [];
  const particles = [];
  const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  function pulseGlow(color){
    glowWash.style.background = `radial-gradient(ellipse at 50% 70%, color-mix(in srgb, ${color} 22%, transparent) 0%, transparent 60%)`;
    glowWash.style.opacity = '1';
    setTimeout(() => { glowWash.style.opacity = '0'; }, 420);
  }

  // a rocket climbing under thrust, decelerating under gravity, until apex -> burst
  class Rocket{
    constructor(){
      this.x = rand(W * 0.12, W * 0.88);
      this.y = H;
      this.targetY = rand(H * 0.16, H * 0.42);
      const dist = this.y - this.targetY;
      // launch velocity tuned so gravity brings it to ~0 velocity near targetY
      this.vy = -Math.sqrt(2 * GRAVITY * dist) * rand(1.0, 1.08);
      this.vx = rand(-6, 6);
      this.color = choice(palette);
      this.type = choice(['peony', 'peony', 'willow', 'ring', 'peony']);
      this.trail = [];
      this.dead = false;
    }
    step(dt){
      this.trail.push({x:this.x, y:this.y});
      if(this.trail.length > 14) this.trail.shift();

      this.vy += GRAVITY * dt * 0.9;
      this.x += this.vx * dt;
      this.y += this.vy * dt;

      if(this.vy >= -6 || this.y <= this.targetY){
        this.explode();
        this.dead = true;
      }
    }
    draw(){
      ctx.lineCap = 'round';
      for(let i = 0; i < this.trail.length; i++){
        const p = this.trail[i];
        const a = i / this.trail.length;
        ctx.beginPath();
        ctx.fillStyle = this.color;
        ctx.globalAlpha = a * 0.8;
        ctx.arc(p.x, p.y, 1.6 * a + 0.4, 0, Math.PI * 2);
        ctx.fill();
      }
      ctx.globalAlpha = 1;
    }
    explode(){
      pulseGlow(this.color);
      spawnFlash(this.x, this.y, this.color);
      spawnBurst(this.x, this.y, this.color, this.type);
    }
  }

  function spawnFlash(x, y, color){
    particles.push({
      x, y, vx:0, vy:0, life:0.18, maxLife:0.18,
      size:26, color, flash:true
    });
  }

  // a single spark: real velocity + gravity + drag, can spawn crackle children on death
  class Spark{
    constructor(x, y, angle, speed, color, opts = {}){
      this.x = x; this.y = y;
      this.vx = Math.cos(angle) * speed;
      this.vy = Math.sin(angle) * speed;
      this.color = color;
      this.gravityMult = opts.gravityMult ?? 1;
      this.dragMult = opts.dragMult ?? 1;
      this.size = opts.size ?? rand(1.6, 2.6);
      this.life = 0;
      this.maxLife = opts.maxLife ?? rand(1.2, 1.8);
      this.crackle = opts.crackle ?? false;
      this.crackled = false;
      this.willow = opts.willow ?? false;
      this.dead = false;
    }
    step(dt){
      this.life += dt;
      if(this.life >= this.maxLife){ this.dead = true; return; }

      const drag = Math.pow(DRAG, this.dragMult) ;
      this.vx = this.vx * drag + WIND() * dt;
      this.vy = this.vy * drag + GRAVITY * this.gravityMult * dt;
      this.x += this.vx * dt;
      this.y += this.vy * dt;

      // crackle: near end of life, a spark can pop into a few tiny short-lived children
      if(this.crackle && !this.crackled && this.life > this.maxLife * 0.55 && Math.random() < 0.045){
        this.crackled = true;
        const n = Math.floor(rand(3, 6));
        for(let i = 0; i < n; i++){
          const a = rand(0, Math.PI * 2);
          const s = rand(20, 55);
          particles.push(new Spark(this.x, this.y, a, s, '#fff3d0', {
            gravityMult: 1.3, dragMult: 1.4, size: rand(1, 1.6), maxLife: rand(0.25, 0.45)
          }));
        }
      }
    }
    draw(){
      const t = this.life / this.maxLife;
      const alpha = t < 0.08 ? t / 0.08 : 1 - Math.pow((t - 0.08) / 0.92, 1.4);
      if(alpha <= 0) return;
      ctx.globalAlpha = Math.max(0, alpha);
      ctx.beginPath();
      ctx.fillStyle = this.color;
      const sz = this.size * (1 - t * 0.4);
      ctx.arc(this.x, this.y, Math.max(0.3, sz), 0, Math.PI * 2);
      ctx.fill();
      ctx.globalAlpha = 1;
    }
  }

  function spawnBurst(x, y, color, type){
    const secondColor = Math.random() > 0.55 ? choice(palette) : color;
    let count, speedMin, speedMax, gravityMult, dragMult, maxLife, crackleChance;

    if(type === 'willow'){
      count = Math.floor(rand(24, 34));
      speedMin = 50; speedMax = 85;
      gravityMult = 1.5; dragMult = 0.7;
      maxLife = rand(2.2, 2.8);
      crackleChance = 0;
    } else if(type === 'ring'){
      count = Math.floor(rand(26, 32));
      speedMin = 95; speedMax = 110;
      gravityMult = 0.9; dragMult = 1.1;
      maxLife = rand(1.3, 1.6);
      crackleChance = 0.2;
    } else {
      count = Math.floor(rand(30, 42));
      speedMin = 55; speedMax = 125;
      gravityMult = 1; dragMult = 1;
      maxLife = rand(1.3, 1.8);
      crackleChance = 0.22;
    }

    for(let i = 0; i < count; i++){
      const angle = type === 'ring'
        ? (Math.PI * 2 * i) / count + rand(-0.02, 0.02)
        : rand(0, Math.PI * 2);
      const speed = rand(speedMin, speedMax);
      const c = i % 3 === 0 ? secondColor : color;
      particles.push(new Spark(x, y, angle, speed, c, {
        gravityMult, dragMult, maxLife: maxLife * rand(0.85, 1.05),
        crackle: Math.random() < crackleChance,
        willow: type === 'willow',
        size: type === 'willow' ? rand(1.2, 1.8) : rand(1.6, 2.6)
      }));
    }
  }

  function launchFirework(){
    rockets.push(new Rocket());
  }

  function scheduleNext(){
    launchFirework();
    if(Math.random() > 0.85) setTimeout(launchFirework, rand(200, 400));
    setTimeout(scheduleNext, rand(1400, 2600));
  }

  // --- main loop ---
  let lastT = performance.now();
  function tick(now){
    const dt = Math.min((now - lastT) / 1000, 0.05);
    lastT = now;

    // motion-blur trail fade: paint a translucent rect instead of hard clear
    ctx.globalCompositeOperation = 'source-over';
    ctx.fillStyle = 'rgba(4, 6, 13, 0.22)';
    ctx.fillRect(0, 0, W, H);

    ctx.globalCompositeOperation = 'lighter';

    for(let i = rockets.length - 1; i >= 0; i--){
      const r = rockets[i];
      r.step(dt);
      r.draw();
      if(r.dead) rockets.splice(i, 1);
    }

    for(let i = particles.length - 1; i >= 0; i--){
      const p = particles[i];
      if(p.flash){
        p.life += dt;
        const t = p.life / p.maxLife;
        if(t >= 1){ particles.splice(i, 1); continue; }
        ctx.globalAlpha = 1 - t;
        const grad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size * (0.3 + t));
        grad.addColorStop(0, '#fff');
        grad.addColorStop(0.4, p.color);
        grad.addColorStop(1, 'transparent');
        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size * (0.3 + t), 0, Math.PI * 2);
        ctx.fill();
        ctx.globalAlpha = 1;
        continue;
      }
      p.step(dt);
      p.draw();
      if(p.dead) particles.splice(i, 1);
    }

    ctx.globalCompositeOperation = 'source-over';
    requestAnimationFrame(tick);
  }

  if(reduceMotion){
    ctx.fillStyle = 'rgba(4,6,13,1)';
    ctx.fillRect(0,0,W,H);
  } else {
    requestAnimationFrame(tick);
    launchFirework();
    setTimeout(launchFirework, 900);
    setTimeout(scheduleNext, 2000);
  }
</script>
